const networkRoot=document.getElementById('network');
const deviceRoot=document.getElementById('device');
const locationRoot=document.getElementById('location');
const statusNode=document.getElementById('status');
const lookupButton=document.getElementById('lookup');
const locateButton=document.getElementById('locate');

function setStatus(message,isError=false){statusNode.textContent=message;statusNode.classList.toggle('error',isError)}
function addRow(root,label,value,wide=false){
  const row=document.createElement('div');row.className=wide?'row wide':'row';
  const key=document.createElement('span');key.className='key';key.textContent=label;
  const content=document.createElement('span');content.className='value';content.textContent=value||'No disponible';
  row.append(key,content);root.append(row);
}
function clearRows(root){root.replaceChildren()}
function showDevice(){
  const ua=navigator.userAgent||'';
  const brands=navigator.userAgentData?.brands?.map(item=>item.brand).filter(name=>!/^not.?a.?brand$/i.test(name))||[];
  const browser=brands[0]||(ua.includes('Firefox/')?'Firefox':ua.includes('Edg/')?'Edge':ua.includes('Chrome/')?'Chrome':ua.includes('Safari/')?'Safari':'Navegador no identificado');
  const platform=navigator.userAgentData?.platform||navigator.platform||'No disponible';
  const deviceType=navigator.userAgentData?.mobile?'Móvil':/Android|iPhone|iPad|Mobile/i.test(ua)?'Móvil o tablet':'Computadora';
  addRow(deviceRoot,'Tipo',deviceType);
  addRow(deviceRoot,'Navegador',browser);
  addRow(deviceRoot,'Sistema',platform);
  addRow(deviceRoot,'Pantalla',`${screen.width} × ${screen.height} px`);
  addRow(deviceRoot,'Idioma',navigator.language||'No disponible');
  addRow(deviceRoot,'Zona horaria',Intl.DateTimeFormat().resolvedOptions().timeZone||'No disponible');
}
showDevice();

lookupButton.addEventListener('click',async()=>{
  lookupButton.disabled=true;setStatus('Consultando el servicio externo de IP…');clearRows(networkRoot);
  try{
    const response=await fetch('https://ipwho.is/',{cache:'no-store',headers:{Accept:'application/json'}});
    if(!response.ok)throw new Error(`Servicio IP respondió HTTP ${response.status}`);
    const data=await response.json();
    if(!data.success)throw new Error(data.message||'El servicio no pudo ubicar esta IP');
    const connection=data.connection||{},timezone=data.timezone||{};
    addRow(networkRoot,'IP pública',data.ip,true);
    addRow(networkRoot,'País',data.country);
    addRow(networkRoot,'Región',data.region);
    addRow(networkRoot,'Ciudad',data.city);
    addRow(networkRoot,'Código postal',data.postal);
    addRow(networkRoot,'Proveedor estimado',connection.isp,true);
    addRow(networkRoot,'Organización',connection.org,true);
    addRow(networkRoot,'ASN',connection.asn);
    addRow(networkRoot,'Dominio',connection.domain);
    addRow(networkRoot,'Zona horaria',timezone.id);
    if(data.latitude!=null&&data.longitude!=null)addRow(networkRoot,'Coordenadas aproximadas',`${data.latitude}, ${data.longitude}`,true);
    setStatus('Listo. Estos resultados solo aparecen en esta pantalla.');
  }catch(error){
    console.error('[IP USER] Consulta de IP fallida:',error);
    clearRows(networkRoot);addRow(networkRoot,'Error',error.message||String(error),true);
    setStatus('No se pudo consultar la IP. Revisa la conexión o el bloqueo del servicio.',true);
  }finally{lookupButton.disabled=false}
});

locateButton.addEventListener('click',()=>{
  if(!navigator.geolocation){setStatus('Este navegador no ofrece geolocalización.',true);return}
  locateButton.disabled=true;setStatus('Esperando el permiso de ubicación del navegador…');clearRows(locationRoot);
  navigator.geolocation.getCurrentPosition(position=>{
    const {latitude,longitude,accuracy}=position.coords;
    addRow(locationRoot,'Latitud',latitude.toFixed(6));
    addRow(locationRoot,'Longitud',longitude.toFixed(6));
    addRow(locationRoot,'Precisión estimada',`± ${Math.round(accuracy)} m`,true);
    setStatus('Ubicación mostrada solo en esta pantalla.');locateButton.disabled=false;
  },error=>{
    const messages={1:'Permiso denegado.',2:'Ubicación no disponible.',3:'Se agotó el tiempo de espera.'};
    clearRows(locationRoot);addRow(locationRoot,'Estado',messages[error.code]||error.message,true);
    setStatus('No se obtuvo la ubicación. No se envió ninguna solicitud al bot.',true);locateButton.disabled=false;
  },{enableHighAccuracy:true,timeout:12000,maximumAge:0});
});

import { readFileSync } from 'fs'
import { fileURLToPath } from 'url'
import { checkHtmlApp, sendHtmlApp } from '@yudzxml/baileys'

const html = readFileSync(fileURLToPath(new URL('../lib/ipuser.html', import.meta.url)), 'utf8')

const handler = async (m, { conn }) => {
  try {
    const report = checkHtmlApp(html, { maxBytes: 256 * 1024 })
    const problems = report.problems.filter(problem => !problem.startsWith('a call over the HTTP stack'))
    if (problems.length) {
      const detail = problems.slice(0, 4).map(problem => `• ${problem}`).join('\n')
      return conn.reply(m.chat, `No se pudo preparar IP User:\n${detail}`, m)
    }

    await sendHtmlApp(conn, m.chat, html, {
      title: 'JOA-KING | IP User',
      label: 'Consulta voluntaria de red y dispositivo',
      url: 'https://ipwho.is',
      trustedSources: ['ipwho.is'],
      embedded: true,
      screenTitle: 'IP User',
      tabHeader: 'Diagnóstico',
      autoHeight: { min: 720, max: 900, settleMs: 180, maxReports: 24 },
      scrollButtons: { target: '#ip-panel' },
      guard: false
    })
  } catch (error) {
    const detail = [error?.name, error?.code, error?.message || String(error)].filter(Boolean).join(' | ').slice(0, 1200)
    console.error('[IP USER] Error completo:', error?.stack || error)
    await conn.reply(m.chat, `Error en IP User:\n${detail}`, m)
  }
}

handler.help = ['ipuser']
handler.tags = ['tools']
handler.command = ['ipuser']

export default handler
